plc_poll.py
# Poll Siemens S7-1200 holding registers via snap7
import snap7
from snap7.util import get_real
import time

plc = snap7.client.Client()
plc.connect("192.168.0.10", 0, 1)

def read_flow_pv():
    data = plc.db_read(db_number=10, start=0, size=4)
    return get_real(data, 0)

while True:
    pv = read_flow_pv()
    print(f"Custody transfer flow: {pv:.2f} m3/h")
    time.sleep(1)

sp_ShiftReport.sql
-- Aggregate custody-transfer totals per shift
CREATE PROCEDURE sp_ShiftProductionReport
  @ShiftDate DATE,
  @ShiftCode VARCHAR(2)
AS
BEGIN
  SELECT TagName,
         SUM(FlowTotal)  AS TotalVolume_m3,
         AVG(Temperature) AS AvgTemp_C
  FROM dbo.CustodyTransferLog
  WHERE CAST(LogTime AS DATE) = @ShiftDate
    AND Shift = @ShiftCode
  GROUP BY TagName;
END;
